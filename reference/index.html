    def process_entity(
    entity: str,
    records: list[Record],
    sink: Sink,
    partition: str,
    run_id: str,
) -> EntityMetrics:
    """Transforma, valida e grava os registros de uma entidade.

    Raises:
        SchemaValidationError: Se o resultado violar o schema (nada é gravado).
    """
    df, stats = transform(entity, records)
    validate(df, SCHEMAS[entity])
    sink.write(entity, partition, df)
    logger.info(
        "etapa concluída",
        extra={
            "run_id": run_id,
            "step": "load",
            "entity": entity,
            "partition": partition,
            "rows_in": stats.rows_in,
            "rows_rejected": stats.rows_rejected,
            "duplicates_dropped": stats.duplicates_dropped,
            "values_coerced": stats.values_coerced,
            "rows_out": len(df),
        },
    )
    return EntityMetrics(entity, stats.rows_in, stats.rows_rejected, len(df))

    def run_session(
    session: Record,
    meeting: Record,
    extractor: Extractor,
    sink: Sink,
    run_id: str,
    skip_existing: bool = False,
) -> RunReport:
    """Processa uma sessão de corrida e seu fim de semana.

    Reexecutar sobrescreve as partições da sessão (idempotente). Com
    ``skip_existing``, sessões já concluídas são puladas.
    """
    session_key = int(session["session_key"])
    meeting_key = int(session["meeting_key"])
    session_part = f"session_key={session_key}"
    meeting_part = f"meeting_key={meeting_key}"
    started = time.monotonic()
    if skip_existing and sink.exists(SESSION_ENTITIES[-1], session_part):
        logger.info(
            "sessão já concluída; pulando",
            extra={"run_id": run_id, "session_key": session_key},
        )
        return RunReport(run_id, session_key, skipped=True)

    metrics = [
        process_entity("meetings", [meeting], sink, meeting_part, run_id),
        process_entity("sessions", [session], sink, session_part, run_id),
    ]
    for entity in SESSION_ENTITIES:
        records = extractor.session_entity(entity, session_key)
        metrics.append(process_entity(entity, records, sink, session_part, run_id))
    logger.info(
        "sessão concluída",
        extra={
            "run_id": run_id,
            "session_key": session_key,
            "duration_s": round(time.monotonic() - started, 3),
            "rows_written": sum(m.rows_written for m in metrics),
        },
    )
    return RunReport(run_id, session_key, metrics=metrics)

    def run_session_key(
    session_key: int,
    extractor: Extractor,
    sink: Sink,
    run_id: str,
    skip_existing: bool = False,
) -> RunReport:
    """Processa uma sessão a partir do ``session_key``.

    Raises:
        PermanentApiError: Se a sessão ou o fim de semana não forem encontrados.
    """
    sessions = extractor.session(session_key)
    if not sessions:
        raise PermanentApiError(f"Sessão {session_key} não encontrada")
    meetings = extractor.meeting(int(sessions[0]["meeting_key"]))
    if not meetings:
        raise PermanentApiError(f"Fim de semana da sessão {session_key} não encontrado")
    return run_session(sessions[0], meetings[0], extractor, sink, run_id, skip_existing)

    def run_year(
    year: int,
    extractor: Extractor,
    sink: Sink,
    run_id: str,
    start: date | None = None,
    end: date | None = None,
    skip_existing: bool = False,
    session_names: Sequence[str] = DEFAULT_SESSION_NAMES,
) -> tuple[list[RunReport], list[int]]:
    """Processa as sessões de ``session_names`` de um ano (ou janela ``start``-``end``).

    Uma sessão que falhe é registrada e não impede as demais.

    Returns:
        Os relatórios das sessões concluídas e os ``session_key`` que falharam.
    """
    reports: list[RunReport] = []
    failed: list[int] = []
    for meeting in extractor.meetings(year, start, end):
        if meeting.get("is_cancelled"):
            continue
        for session_name in session_names:
            sessions = extractor.sessions_named(
                int(meeting["meeting_key"]), session_name
            )
            for session in sessions:
                try:
                    reports.append(
                        run_session(
                            session, meeting, extractor, sink, run_id, skip_existing
                        )
                    )
                except _RECOVERABLE:
                    logger.exception(
                        "falha ao processar sessão",
                        extra={
                            "run_id": run_id,
                            "session_key": session["session_key"],
                        },
                    )
                    failed.append(int(session["session_key"]))
    return reports, failed

    def build_parser() -> argparse.ArgumentParser:
    """Cria o parser de argumentos."""
    parser = argparse.ArgumentParser(description="Pipeline de corridas da OpenF1")
    target = parser.add_mutually_exclusive_group(required=True)
    target.add_argument("--year", type=int, help="Ano a processar")

    )

    )
    return parser

    def main(
    argv: Sequence[str] | None = None,
    extractor: Extractor | None = None,
    sink: Sink | None = None,
) -> int:
    """Executa o pipeline; retorna 0 em sucesso e 1 se houve falhas."""
    args = build_parser().parse_args(argv)
    if (args.start or args.end) and args.year is None:
        build_parser().error("--start/--end exigem --year")
    configure_logging()
    settings = Settings.from_env()
    extractor = extractor or Extractor(OpenF1Client.from_settings(settings))
    sink = sink or ParquetSink(settings.data_dir)
    run_id = new_run_id()
    started = time.monotonic()
    logger.info("execução iniciada", extra={"run_id": run_id, "cli_args": vars(args)})
    try:
        if args.session_key is not None:
            reports = [
                run_session_key(
                    args.session_key, extractor, sink, run_id, args.skip_existing
                )
            ]
            failed: list[int] = []
        else:
            reports, failed = run_year(
                args.year,
                extractor,
                sink,
                run_id,
                args.start,
                args.end,
                args.skip_existing,
                args.session_names or DEFAULT_SESSION_NAMES,
            )
    except (TransientApiError, PermanentApiError, SchemaValidationError):
        logger.exception("execução abortada", extra={"run_id": run_id})
        return 1
    logger.info(
        "execução finalizada",
        extra={
            "run_id": run_id,
            "duration_s": round(time.monotonic() - started, 3),
            "sessions_ok": len(reports),
            "sessions_failed": failed,
        },
    )
    return 1 if failed else 0

    class Extractor:
    """Consulta a API com retry para falhas transitórias."""

    def __init__(
        self,
        client: ApiClient,
        attempts: int = 5,
        base_delay_s: float = 2.0,
        sleep: Callable[[float], None] = time.sleep,
    ) -> None:
        self._client = client
        self._attempts = attempts
        self._base_delay_s = base_delay_s
        self._sleep = sleep

    def _get(self, endpoint: str, params: Params) -> list[Record]:
        return with_retry(
            lambda: self._client.get(endpoint, params),
            attempts=self._attempts,
            base_delay_s=self._base_delay_s,
            sleep=self._sleep,
        )

    def meetings(
        self, year: int, start: date | None = None, end: date | None = None
    ) -> list[Record]:
        """Fins de semana do ano, opcionalmente dentro da janela ``start``-``end``."""
        params: dict[str, object] = {"year": year}
        if start:
            params["date_start>="] = start.isoformat()
        if end:
            params["date_end<="] = f"{end.isoformat()}T23:59:59"
        return self._get("meetings", params)

    def meeting(self, meeting_key: int) -> list[Record]:
        """Um fim de semana pelo ``meeting_key``."""
        return self._get("meetings", {"meeting_key": meeting_key})

    def sessions_named(self, meeting_key: int, session_name: str) -> list[Record]:
        """Sessões de um fim de semana com o ``session_name`` dado (ex.: ``Race``)."""
        return self._get(
            "sessions", {"meeting_key": meeting_key, "session_name": session_name}
        )

    def session(self, session_key: int) -> list[Record]:
        """Uma sessão pelo ``session_key``."""
        return self._get("sessions", {"session_key": session_key})

    def session_entity(self, entity: str, session_key: int) -> list[Record]:
        """Registros de uma entidade de sessão (ver ``SESSION_ENTITIES``).

        Raises:
            ValueError: Se ``entity`` não for uma entidade de sessão conhecida.
        """
        if entity not in SESSION_ENTITIES:
            raise ValueError(f"Entidade desconhecida: {entity}")
        return self._get(entity, {"session_key": session_key})

    def with_retry[T](
    fn: Callable[[], T],
    *,
    attempts: int = 3,
    base_delay_s: float = 1.0,
    sleep: Callable[[float], None] = time.sleep,
) -> T:
    """Executa ``fn`` repetindo apenas em ``TransientApiError``.

    Erros permanentes e quaisquer outras exceções propagam imediatamente.

    Raises:
        TransientApiError: Se todas as tentativas falharem.
        ValueError: Se ``attempts`` for menor que 1.
    """
    if attempts < 1:
        raise ValueError("attempts deve ser >= 1")
    for attempt in range(1, attempts + 1):
        try:
            return fn()
        except TransientApiError:
            if attempt == attempts:
                raise
            delay = base_delay_s * 2 ** (attempt - 1)
            logger.warning(
                "falha transitória; nova tentativa",
                extra={"attempt": attempt, "delay_s": delay},
            )
            sleep(delay)
    raise AssertionError("inalcançável")  # pragma: no cover

    def transform(
    entity: str, records: list[Record]
) -> tuple[pd.DataFrame, TransformStats]:
    """Converte registros brutos de uma entidade no schema de carga.

    Aplica tipagem explícita, descarta linhas com chave nula e duplicados
    (mantém a última) e devolve as contagens do que foi afetado.

    Raises:
        KeyError: Se ``entity`` não tiver schema.
    """
    schema = SCHEMAS[entity]
    raw = pd.DataFrame.from_records(records)
    if entity == "session_result":
        raw = _add_gap_columns(raw)
    typed, coerced = conform(raw, schema)
    keyed = drop_null_keys(typed, schema.key)
    unique = drop_duplicates_by_key(keyed, schema.key)
    stats = TransformStats(
        rows_in=len(typed),
        rows_rejected=len(typed) - len(keyed),
        duplicates_dropped=len(keyed) - len(unique),
        values_coerced=coerced,
    )
    return unique, stats

    def conform(df: pd.DataFrame, schema: EntitySchema) -> tuple[pd.DataFrame, int]:
    """Aplica os tipos do schema, mantendo só as colunas dele (ausentes viram nulas).

    Returns:
        O DataFrame tipado e a quantidade de valores não nulos que não puderam
        ser convertidos e viraram nulos.
    """
    out = pd.DataFrame(index=df.index)
    coerced = 0
    for column, kind in schema.columns.items():
        source = df[column] if column in df.columns else pd.Series(None, index=df.index)
        typed = _cast(source, kind)
        coerced += int((typed.isna() & source.notna()).sum())
        out[column] = typed
    return out, coerced

    def validate(df: pd.DataFrame, schema: EntitySchema) -> None:
    """Falha rápido se o DataFrame violar o schema.

    Verifica colunas presentes, tipos, chave sem nulos e chave única.

    Raises:
        SchemaValidationError: Com a lista de todas as violações encontradas.
    """
    errors: list[str] = []
    missing = [c for c in schema.columns if c not in df.columns]
    if missing:
        errors.append(f"colunas ausentes: {missing}")
    for column, kind in schema.columns.items():
        if column in df.columns and not _DTYPE_CHECKS[kind](df[column].dtype):
            errors.append(
                f"tipo inesperado em {column}: {df[column].dtype} (esperado {kind})"
            )
    if not missing:
        null_keys = [c for c in schema.key if df[c].isna().any()]
        if null_keys:
            errors.append(f"chave com nulos: {null_keys}")
        duplicated = int(df.duplicated(subset=list(schema.key)).sum())
        if duplicated:
            errors.append(f"{duplicated} linhas com chave duplicada {schema.key}")
    if errors:
        raise SchemaValidationError(f"{schema.name}: " + "; ".join(errors))

    class OpenF1Client:

    def __init__(
        self,
        base_url: str,
        timeout_s: float,
        rate_limit_delay_s: float = 0.0,
        session: requests.Session | None = None,
        sleep: Callable[[float], None] = time.sleep,
    ) -> None:
        self._base_url = base_url.rstrip("/")
        self._timeout_s = timeout_s
        self._delay_s = rate_limit_delay_s
        self._session = session or requests.Session()
        self._sleep = sleep

    @classmethod
    def from_settings(cls, settings: Settings) -> "OpenF1Client":
        """Cria o cliente a partir das configurações."""
        return cls(
            settings.base_url, settings.api_timeout_s, settings.rate_limit_delay_s
        )

    def get(self, endpoint: str, params: Params | None = None) -> list[Record]:
        """Consulta um endpoint e retorna a lista de registros.

        Raises:
            TransientApiError: Timeout, erro de conexão, 429 ou 5xx.
            PermanentApiError: Demais 4xx ou corpo que não seja uma lista JSON.
                Um 404 "sem resultados" não é erro: retorna ``[]``.
        """
        url = f"{self._base_url}/{endpoint}"
        query = build_query(params)
        if query:
            url = f"{url}?{query}"
        if self._delay_s:
            self._sleep(self._delay_s)
        try:
            response = self._session.get(url, timeout=self._timeout_s)
        except (requests.Timeout, requests.ConnectionError) as exc:
            raise TransientApiError(f"Falha de rede em {endpoint}: {exc}") from exc
        status = response.status_code
        if status == 429 or status >= 500:
            raise TransientApiError(f"HTTP {status} em {endpoint}")
        if status == 404 and NO_RESULTS in response.text:
            return []
        if status >= 400:
            raise PermanentApiError(
                f"HTTP {status} em {endpoint}: {response.text[:200]}"
            )
        try:
            data = response.json()
        except ValueError as exc:
            raise PermanentApiError(f"Resposta não é JSON em {endpoint}") from exc
        if not isinstance(data, list):
            raise PermanentApiError(f"Resposta inesperada em {endpoint}: {type(data)}")
        logger.debug(
            "consulta concluída", extra={"endpoint": endpoint, "rows": len(data)}
        )
        return data

    def build_query(params: Params | None) -> str:
    """Monta a query string no formato aceito pela OpenF1.

    Chaves terminadas em operador (``date>=``) viram ``date>=valor``; as demais
    viram ``chave=valor``. Valores em lista geram parâmetros repetidos.
    """
    parts: list[str] = []
    for key, raw in (params or {}).items():
        values = raw if isinstance(raw, list | tuple) else [raw]
        for value in values:
            separator = "" if key.endswith(OPERATORS) else "="
            parts.append(f"{key}{separator}{_format_value(value)}")
    return "&".join(parts)

    class ParquetSink:
    """Grava ``<base_dir>/<entity>/<partition>/part.parquet``.

    A escrita vai para um arquivo temporário que substitui o definitivo de uma
    só vez; regravar a mesma partição é idempotente.
    """

    def __init__(self, base_dir: Path) -> None:
        self._base_dir = base_dir

    def _path(self, entity: str, partition: str) -> Path:
        return self._base_dir / entity / partition / FILE_NAME

    def write(self, entity: str, partition: str, df: pd.DataFrame) -> Path:
        """Grava a partição, substituindo o conteúdo anterior."""
        target = self._path(entity, partition)
        target.parent.mkdir(parents=True, exist_ok=True)
        tmp = target.with_suffix(".parquet.tmp")
        try:
            df.to_parquet(tmp, index=False)
            os.replace(tmp, target)
        finally:
            tmp.unlink(missing_ok=True)
        logger.debug("partição gravada", extra={"path": str(target), "rows": len(df)})
        return target

    def exists(self, entity: str, partition: str) -> bool:
        """Indica se a partição já foi gravada."""
        return self._path(entity, partition).is_file()
